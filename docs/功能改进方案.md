        "success": 0,

"""
mining_state.py —— 策略挖掘状态管理（避免循环导入）
"""

_mining_status = {
    "running": False,
    "progress": None,
    "result": None,
    "stop_requested": False
}


def get_status():
    """获取挖掘状态"""
    return _mining_status


def request_stop():
    """请求停止挖掘"""
    _mining_status["stop_requested"] = True


def reset():
    """重置状态"""
    _mining_status.update({
        "running": False,
        "stop_requested": False,
        "progress": None,
        "result": None
    })

from strategy.mining_state import get_status, request_stop, reset

@strategy_bp.route("/mine", methods=["POST"])
def start_mining():
    """触发策略挖掘"""
    status = get_status()
    if status["running"]:
        return jsonify({"success": False, "error": "Mining already in progress"}), 409
    
    # 重置状态
    reset()
    status["running"] = True
    
    body = request.get_json(silent=True) or {}
    trade_date = body.get("trade_date") or _get_latest_trade_date()
    
    def _run():
        try:
            from strategy.miner import mine_strategies
            saved = mine_strategies(trade_date)
            status["result"] = saved
            status["progress"] = "done"
        except Exception as e:
            status["progress"] = "error"
            status["result"] = str(e)
        finally:
            status["running"] = False
    
    threading.Thread(target=_run, daemon=True).start()
    return jsonify({"success": True, "data": {"status": "started"}}, 202)

@strategy_bp.route("/mine/stop", methods=["POST"])
def stop_mining():
    """停止策略挖掘 POST /api/strategy/mine/stop"""
    status = get_status()
    if not status["running"]:
        return jsonify({"success": False, "error": "Mining not in progress"}), 400
    
    request_stop()
    return jsonify({"success": True, "message": "Stop requested"})

@strategy_bp.route("/mine/status", methods=["GET"])
def mining_status():
    """查询挖掘进度"""
    return jsonify({"success": True, "data": get_status()})

@backtest_bp.route("/compare", methods=["GET"])
def compare_results():
    """对比多个回测结果 GET /api/backtest/compare?ids=1,2,3"""
    ids = request.args.get("ids", "").split(",")
    results = []
    
    for rid in ids:
        try:
            rid_int = int(rid)
            detail = build_result_detail(rid_int)
            if detail:
                summary = detail["summary"].copy()
                summary["result_id"] = rid_int  # 确保包含ID
                results.append(summary)
        except (ValueError, TypeError):
            continue
    
    return jsonify({
        "success": True, 
        "data": results,
        "count": len(results)
    })

// 回测结果列表增加复选框
let selectedCompareIds = new Set();

function renderResultsList(results) {

                       onchange="toggleCompare(${r.result_id}, this.checked)"></td>
            <td>${r.rule_name}</td>
            <td>${r.annual_return}%</td>
            <td>${r.win_rate}%</td>
            <td>${r.sharpe_ratio}</td>
            <td>${r.max_drawdown}%</td>
        </tr>`;
    });
    
    html += `</table>
        <button class="btn btn-primary" id="compareBtn" 
                onclick="compareSelected()" 
                ${selectedCompareIds.size < 2 ? 'disabled' : ''}>
            对比选中 (${selectedCompareIds.size})
        </button>`;
    
    return html;
}

function toggleCompare(resultId, checked) {
    if (checked) {
        selectedCompareIds.add(resultId);
    } else {
        selectedCompareIds.delete(resultId);
    }
    
    // 更新按钮状态
    const btn = document.getElementById('compareBtn');
    btn.disabled = selectedCompareIds.size < 2;
    btn.textContent = `对比选中 (${selectedCompareIds.size})`;
}

async function compareSelected() {
    if (selectedCompareIds.size < 2) {
        alert('请至少选择2条回测结果进行对比');
        return;
    }
    
    const ids = Array.from(selectedCompareIds).join(',');
    const resp = await fetchJSON(API + `/backtest/compare?ids=${ids}`);
    showCompareModal(resp.data);
}

function showCompareModal(results) {
    // 找出每项指标的最优值
    const bestValues = {
        annual_return: Math.max(...results.map(r => r.annual_return)),
        win_rate: Math.max(...results.map(r => r.win_rate)),
        sharpe_ratio: Math.max(...results.map(r => r.sharpe_ratio)),
        max_drawdown: Math.min(...results.map(r => r.max_drawdown))  // 回撤越小越好
    };
    
    // 渲染对比表格
    let html = '<table><tr><th>指标</th>';
    results.forEach(r => {
        html += `<th>${r.rule_name}</th>`;
    });
    html += '</tr>';
    
    // 年化收益
    html += '<tr><td>年化收益</td>';
    results.forEach(r => {
        const isBest = r.annual_return === bestValues.annual_return;
        html += `<td class="${isBest ? 'best-value' : ''}">${r.annual_return ?? '--'}%</td>`;
    });
    html += '</tr>';
    
    // 胜率
    html += '<tr><td>胜率</td>';
    results.forEach(r => {
        const isBest = r.win_rate === bestValues.win_rate;
        html += `<td class="${isBest ? 'best-value' : ''}">${r.win_rate ?? '--'}%</td>`;
    });
    html += '</tr>';
    
    // 夏普比率
    html += '<tr><td>夏普比率</td>';
    results.forEach(r => {
        const isBest = r.sharpe_ratio === bestValues.sharpe_ratio;
        html += `<td class="${isBest ? 'best-value' : ''}">${r.sharpe_ratio ?? '--'}</td>`;
    });
    html += '</tr>';
    
    // 最大回撤
    html += '<tr><td>最大回撤</td>';
    results.forEach(r => {
        const isBest = r.max_drawdown === bestValues.max_drawdown;
        html += `<td class="${isBest ? 'best-value' : ''}">${r.max_drawdown ?? '--'}%</td>`;
    });
    html += '</tr></table>';
    
    showModal(html);
}

def save_rule_version(rule_id: int) -> int:
    """保存当前规则为历史版本，version从1开始自增"""
    with get_conn() as conn:
        rule = conn.execute(
            "SELECT * FROM strategy_rules WHERE id = ?", 
            (rule_id,)
        ).fetchone()
        
        if not rule:
            raise ValueError(f"Rule {rule_id} not found")
        
        # 获取当前最大版本号
        max_ver_row = conn.execute(
            "SELECT MAX(version) as v FROM strategy_rule_versions WHERE rule_id = ?",
            (rule_id,)
        ).fetchone()
        
        next_version = (max_ver_row["v"] or 0) + 1
        
        conn.execute("""
            INSERT INTO strategy_rule_versions
                (rule_id, version, conditions, sell_conditions, 
                 holding_min, holding_max, fitness, annual_return, 
                 win_rate, sharpe_ratio, max_drawdown)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        """, (
            rule_id, next_version, rule["conditions"], rule["sell_conditions"],
            rule["holding_min"], rule["holding_max"], rule["fitness"],
            rule["annual_return"], rule["win_rate"], rule["sharpe_ratio"],
            rule["max_drawdown"]
        ))
        
        return next_version


def get_rule_versions(rule_id: int) -> list:
    """获取某规则所有历史版本"""
    with get_conn() as conn:
        rows = conn.execute("""
            SELECT * FROM strategy_rule_versions 
            WHERE rule_id = ? 
            ORDER BY version DESC
        """, (rule_id,)).fetchall()
        return [dict(r) for r in rows]


def rollback_rule(rule_id: int, version: int) -> bool:
    """回滚到指定版本，回滚前自动保存当前版本"""
    with get_conn() as conn:
        # 验证版本存在且属于该rule_id
        ver = conn.execute(
            "SELECT * FROM strategy_rule_versions WHERE rule_id = ? AND version = ?",
            (rule_id, version)
        ).fetchone()
        
        if not ver:
            return False
        
        # 回滚前先保存当前版本（防止误操作）
        save_rule_version(rule_id)
        
        # 执行回滚（不修改created_at等元数据）
        conn.execute("""
            UPDATE strategy_rules SET
                conditions = ?, 
                sell_conditions = ?,
                holding_min = ?, 
                holding_max = ?,
                fitness = ?, 
                annual_return = ?, 
                win_rate = ?,
                sharpe_ratio = ?, 
                max_drawdown = ?,
                updated_at = datetime('now', 'localtime')
            WHERE id = ?
        """, (
            ver["conditions"], ver["sell_conditions"],
            ver["holding_min"], ver["holding_max"],
            ver["fitness"], ver["annual_return"], ver["win_rate"],
            ver["sharpe_ratio"], ver["max_drawdown"],
            rule_id
        ))
        return True


def save_rule(rule: dict) -> int:
    """插入或更新一条规则（按rule_name去重）"""
    with get_conn() as conn:
        existing = conn.execute(
            "SELECT id FROM strategy_rules WHERE rule_name = ?", 
            (rule["rule_name"],)
        ).fetchone()
        
        # 仅在更新已有规则时保存版本
        if existing:
            save_rule_version(existing["id"])
        
        # ... 后续保存逻辑
        return existing["id"] if existing else cur.lastrowid

def sync_one_stock_with_retry_and_timeout(
    code: str, 
    start_date: str = HISTORY_START,
    end_date: str = None, 
    verbose: bool = False,
    max_retries: int = 2,
    timeout: float = 30.0
) -> bool:
    """带重试和超时的单股票同步（合并超时/重试逻辑）"""
    for attempt in range(max_retries + 1):
        try:
            with ThreadPoolExecutor(max_workers=1) as ex:
                fut = ex.submit(
                    sync_one_stock, 
                    code, start_date, end_date, verbose, 
                    auto_login=(attempt == 0)  # 首次用外部会话，重试时自动login
                )
                ok = fut.result(timeout=timeout)
                
                if ok:
                    return True
                    
                if attempt < max_retries:
                    wait = 2 ** attempt
                    if verbose:
                        print(f"  [{code}] 第{attempt+1}次失败，{wait}秒后重试...")
                    time.sleep(wait)
                    
        except FutureTimeoutError:
            if attempt < max_retries:
                wait = 2 ** attempt
                if verbose:
                    print(f"  [{code}] 超时，{wait}秒后重试...")
                time.sleep(wait)
                
        except Exception as e:
            if attempt < max_retries:
                wait = 2 ** attempt
                if verbose:
                    print(f"  [{code}] 异常: {e}，{wait}秒后重试...")
                time.sleep(wait)
    
    return False


def daily_sync(verbose=True, progress_callback=None):
    """每日同步入口"""

            continue

            _sync_stats["success"] += 1
        else:
            _sync_stats["failed"] += 1
        
        if progress_callback:
            progress_callback(
                _sync_stats["success"] + _sync_stats["failed"],
                _sync_stats["total"],
                _sync_stats["success"],

_sync_retry_count = 0
_MAX_SYNC_RETRIES = 1  # 只重试1次

def run_sync_blocking(progress_callback=None):
    """执行同步，失败后自动重试一次"""
    global _sync_retry_count
    
    try:
        daily_sync(verbose=True, progress_callback=progress_callback)
        SYNC_STATUS["last_result"] = "同步成功"
        _sync_retry_count = 0  # 成功后重置
    except Exception as e:
        SYNC_STATUS["last_result"] = f"同步失败: {e}"
        
        # 只重试一次
        if _sync_retry_count < _MAX_SYNC_RETRIES:
            _sync_retry_count += 1
            
            def retry_later():
                time.sleep(1800)  # 30分钟
                SYNC_STATUS["last_result"] = "自动重试中..."
                try:
                    daily_sync(verbose=True)
                    SYNC_STATUS["last_result"] = "重试成功"
                    _sync_retry_count = 0
                except Exception as e2:
                    SYNC_STATUS["last_result"] = f"重试失败: {e2}"
                    _sync_retry_count = _MAX_SYNC_RETRIES  # 标记已达上限
            
            threading.Thread(target=retry_later, daemon=True).start()
        else:
            SYNC_STATUS["last_result"] += " (已达重试上限)"
